cmake_minimum_required(VERSION 3.16)
project(dds_bench)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
add_executable(dds_bench benchmark.cpp)
ament_target_dependencies(dds_bench rclcpp std_msgs)
add_executable(dds_bench_contract_test contract_test.cpp)
target_compile_options(dds_bench_contract_test PRIVATE -UNDEBUG)
set(MDDS_KH_RMW_LIBRARY "" CACHE FILEPATH "KH RMW shared library for direct binding")
set(MDDS_KH_BRIDGE_LIBRARY "" CACHE FILEPATH "Matching KH GN bridge")
set(MDDS_KH_SYSTEM_LINK_DIR "" CACHE PATH "Board platform libraries used for cross-link validation")
if(MDDS_KH_RMW_LIBRARY AND MDDS_KH_BRIDGE_LIBRARY)
  add_library(kh_rmw SHARED IMPORTED)
  set_target_properties(kh_rmw PROPERTIES IMPORTED_LOCATION "${MDDS_KH_RMW_LIBRARY}")
  add_library(kh_bridge SHARED IMPORTED)
  set_target_properties(kh_bridge PROPERTIES IMPORTED_LOCATION "${MDDS_KH_BRIDGE_LIBRARY}")
  add_executable(dds_bench_kh benchmark.cpp)
  target_link_libraries(dds_bench_kh -Wl,--no-as-needed kh_rmw kh_bridge -Wl,--as-needed)
  ament_target_dependencies(dds_bench_kh rclcpp std_msgs)
  get_filename_component(kh_dir "${MDDS_KH_BRIDGE_LIBRARY}" DIRECTORY)
  target_link_options(dds_bench_kh PRIVATE "-Wl,-rpath-link,${kh_dir}" "-Wl,-rpath-link,${MDDS_KH_SYSTEM_LINK_DIR}")
endif()
